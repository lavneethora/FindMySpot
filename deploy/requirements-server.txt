# Runtime dependencies for a deployed pipeline. Deliberately NOT requirements.txt.
#
# ultralytics and torch are absent, and that is the whole point. Detections are
# precomputed into data/cache, and parktech/pipeline.py only imports YOLO inside
# _load_model(), which runs on a cache miss. With a complete cache for the frames
# being replayed, that never happens.
#
# Dropping them takes the image from multiple gigabytes to a few hundred
# megabytes, removes the need for a GPU, and removes the slowest part of the
# cold start. If the cache is incomplete the server will try to import YOLO and
# fail loudly, which is the correct outcome: silently falling back to live
# inference on a small VM would be far worse than a clear error.
fastapi>=0.115.0
uvicorn[standard]>=0.32.0
websockets>=13.0
psycopg[binary]>=3.2.0
numpy>=1.26.0

# headless, not opencv-python: the server has no display, and the GUI build
# pulls in X11 libraries that are dead weight in a container.
opencv-python-headless>=4.10.0
